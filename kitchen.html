// 4. 取得看板資料與偵測新訂單 async function fetchKitchenData() { try { const res = await fetch(`${GAS_URL}?action=getKitchenOrders`); const data = await res.json(); isOpenState = data.isOpen; updateStatusUI(isOpenState); const orders = data.orders || []; // 檢查新訂單並觸發語音/出單 if (!isFirstLoad) { orders.forEach(order => { if (!existingOrderIds.has(order.orderId)) { // 💡 已修改：僅播報「你有新訂單！訂單編號 #X」 const cleanOrderId = order.orderId.replace('#', ''); speakText(`你有新訂單！訂單編號 ${cleanOrderId}`); if (printCharacteristic) { printOrder(order); } } }); } // 更新目前單號列表 existingOrderIds = new Set(orders.map(o => o.orderId)); isFirstLoad = false; renderOrders(orders); if (data.menu) { menuList = data.menu; renderMenuMgmt(); } } catch (err) { console.error("載入失敗:", err); } }
